
Sub SortIt()
' Paul Beverley - Version 09.08.26
' Sorts the selected text (or the whole text)

anyCase = False
promptForWholeText = True

If Selection.End = Selection.Start Then
  Set rng = ActiveDocument.Content
  If promptForWholeText = True Then
    myResponse = MsgBox("Sort the WHOLE file?", _
         vbQuestion + vbYesNo)
    If myResponse = vbNo Then Exit Sub
  End If
Else
  Set rng = Selection.Range.Duplicate
End If
rng.Sort ExcludeHeader:=False, FieldNumber:="Paragraphs", _
     CaseSensitive:=Not (anyCase), SortOrder:=wdSortOrderAscending, _
    SortFieldType:=wdSortFieldAlphanumeric
' Remove blank lines at the top
  rng.InsertAfter Text:=vbCr
  rng.Collapse wdCollapseStart
  rng.MoveEnd wdParagraph, 1
  Do While rng.Text = vbCr
    rng.Paragraphs(1).Range.Delete
    rng.MoveEnd wdParagraph, 1
  Loop
End Sub
