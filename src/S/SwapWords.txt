
Sub SwapWords()
' Paul Beverley - Version 17.07.26
' Swaps adjacent words

Set rng = Selection.Range.Duplicate
rng.Expand wdWord
wd1 = rng
rng.MoveEnd wdWord, 1
wd2 = rng.Words(2)
If Right(wd2, 1) = " " Then
  rng.Text = wd2 & wd1
Else
  rng.Text = wd2 & " " & Trim(wd1)
End If
End Sub
