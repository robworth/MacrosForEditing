
Sub TrackAcceptMoveOn()
' Paul Beverley - Version 02.10.26
' Accepts change and moves to the next

addHighlight = False
myColour = wdYellow
addColour = False
myFontColour = wdRed

Set doc = ActiveDocument
Set rng = Selection.Range
' If a tracked change is selected, accept it and move on
If rng.Revisions.Count > 0 Then
  Set rev = rng.Revisions(1)
  If addHighlight = True Then _
       rng.HighlightColorIndex = myColour
  If addColour = True Then _
       rng.Font.ColorIndex = myFontColour
  curPos = rev.Range.End   ' remember position before accepting
  rev.Accept
Else
  ' If nothing (or no change) is selected, start from current cursor
  curPos = rng.End
End If

' Find the next revision after curPos
myFound = False
For Each rev In doc.Revisions
  If rev.Range.Start >= curPos Then
    rev.Range.Select
    myFound = True
    Exit For
  End If
Next rev

' If none ahead, wrap to the first revision (if any)
If Not myFound Then
  If doc.Revisions.Count > 0 Then
    doc.Revisions(1).Range.Select
  Else
    MsgBox "No tracked changes found.", vbInformation
  End If
End If
End Sub
